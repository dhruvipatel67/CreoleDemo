CREATE TABLE Users (
    UserID int NOT NULL AUTO_INCREMENT,
    LastName varchar(255),
    FirstName varchar(255),
    Address varchar(255),
    City varchar(255) DEFAULT 'Ahmedabad',
    State varchar(255) DEFAULT 'India',
    Age int,
    CHECK (Age>=18)
);

CREATE TABLE Orders (
    OrderID int NOT NULL,
    OrderNumber int NOT NULL,
    OrderDate date DEFAULT CURRENT_DATE(),
    UserID int NOT NULL,
    PRIMARY KEY (OrderID),
    FOREIGN KEY (UserID ) REFERENCES Users(UserID)
);

CREATE TABLE Customers (
    OrderID int NOT NULL,
    CustomerID int NOT NULL,
    CustomerName varchar(40),
    ContactName varchar(20),
    Adddress varchar(100),
    City varchar(20),
    PostalCode varchar(10),
    Country varchar(20)
);

CREATE TABLE BankDeatils (
    HolderID int NOT NULL,
    Holdername varchar(35),
    BankBranch varchar(50),
    Address varchar(200),
    City varchar(25)
);

CREATE TABLE Products (
    ProductID int NOT NULL,
    ProductName varchar(255),
    SupplierID int NOT NULL,
    CategoryID int(20),
    Unit varchar(100),
    Price DECIMAL(6,3)
);

CREATE TABLE Employees (
   Employeeid INT PRIMARY KEY,
   Name VARCHAR (25) ,
   Email VARCHAR (40),
   Department VARCHAR(30)
);

CREATE TABLE OrderDetails (
    OrderDetailID INT PRIMARY KEY,
    OrderID INT PRIMARY KEY,
    ProductID int NOT NULL,
    Quantity int NOT NULL
);

---------------------------------ALIASES--------------------------------------

FROM Customers;

SELECT CustomerID AS ID, CustomerName AS Customer
FROM Customers;

FROM Customers AS c, Orders AS o
WHERE c.CustomerName='Around the Horn' AND c.CustomerID=o.CustomerID;

FROM Customers, Orders

---------------------------------GROUP BY EXAMPLES--------------------------------------

--------------------------------HAVING EXAMPLES-------------------------------------
SELECT COUNT(CustomerID), Country ,CustomerName
FROM Customers
GROUP BY Country
HAVING COUNT(CustomerID) < 5;

FROM Customers
GROUP BY CustomerName
HAVING COUNT(CustomerID) < 5;
